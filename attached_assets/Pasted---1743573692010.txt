from flask_sqlalchemy import SQLAlchemy
from datetime import datetime

db = SQLAlchemy()

class User(db.Model):
    __tablename__ = 'users' # تحديد اسم الجدول صراحةً
    email = db.Column(db.String(100), primary_key=True)
    password = db.Column(db.String(255), nullable=False) # مشفر بـ bcrypt
    role = db.Column(db.String(50), nullable=False)
    preferences = db.Column(db.JSON, default={"language": "ar", "theme": "light", "expo_push_token": None})
    phone = db.Column(db.String(20)) # يجب تشفيره
    wallet_address = db.Column(db.String(42))

class Task(db.Model):
    __tablename__ = 'tasks'
    id = db.Column(db.Integer, primary_key=True, autoincrement=True)
    student_name = db.Column(db.String(100), nullable=False)
    task_content = db.Column(db.Text, nullable=False) # يمكن تشفيره

    feedback = db.Column(db.Text)

    evaluator_email = db.Column(db.String(100), db.ForeignKey('users.email'))

    blockchain_hash = db.Column(db.String(66))
    global_rank = db.Column(db.Integer)

def grok_vision_analyze(content, criteria=None, is_chat=False, language="ar"):
    # ... (الكود كما هو في الرد السابق، مع تحسينات محتملة)
    # ... (تأكد من التعامل مع الأخطاء إذا فشلت نماذج الـ AI في التحميل)
    try:
        if is_chat:
            if "كيف حالك" in content or "how are you" in content:
                return "أنا Grok، نجم التكنولوجيا! جاهز لأذهلك بقدراتي اليوم!"
            elif "ساعدني" in content or "help me" in content:
                return "أنا هنا لأكون نجمك المساعد! ما تحتاجه؟ تحليل؟ نصيحة؟ أم إبهار؟"
            else:
                return f"Grok يقول: '{content}' يستحق ردًا مذهلاً! كيف يمكنني جعله حديث الزمان؟"

        # تحليل متعدد الوسائط
        sentiment = sentiment_analyzer(content)[0]
        strengths = "الإبداع والأصالة" if "مثال" in content else "الوضوح والتركيز"
        weaknesses = "قلة الأدلة" if len(content.split()) < 50 else "التكرار"
        recommendations = (
            "أضف رسومًا بيانية لإبهار المقيمين" if "بيانات" in content else
            "جرب تقنيات جديدة مثل الواقع المعزز لجعل عملك حديث الزمان!"
        )
        feedback = (
            f"تحليل Grok Vision:\n"
            f"- نقاط القوة: {strengths}\n"
            f"- نقاط الضعف: {weaknesses}\n"
            f"- تحليل عاطفي: {sentiment['label']} (الثقة: {sentiment['score']:.2f})\n"
            f"- توصيات ثورية: {recommendations}"
        )
        if language != "ar":
            # تأكد من أن النموذج يدعم اللغات المطلوبة
            translated_feedback = translator(feedback, src_lang="ar", tgt_lang=language)
            if translated_feedback:
                 feedback = translated_feedback[0]['translation_text']
            else:
                 logging.warning(f"Translation failed for language: {language}")

        return feedback
    except Exception as e:
        logging.error(f"Error during Grok analysis: {e}")
        return "حدث خطأ أثناء التحليل البطولي!"

@bp.route('/analyze', methods=['POST'])
@jwt_required()
def analyze_task():
    task_id = request.form.get("task_id")
    language = request.form.get("language", "ar")
    file = request.files.get("file")
    task = Task.query.get(task_id)
    if not task:
        return jsonify({"error": "Task not found"}), 404

    try:
        feedback, sentiment = analyze_with_advanced_ai(task_id, task.task_content, task.criteria, file, language)
        task.feedback = feedback
        task.sentiment = sentiment['label']
        # تحديث الترتيب العالمي هنا بعد التحليل
        score_value = task.plagiarism_score if task.plagiarism_score is not None else 0.0
        global_score = (1 - score_value) * 100 + (50 if task.approved else 0)
        task.global_rank = int(global_score)

        db.session.commit() # حفظ التغييرات
        return jsonify({"feedback": feedback, "sentiment": sentiment})
    except Exception as e:
        logging.error(f"Error in /analyze endpoint for task {task_id}: {e}")
        return jsonify({"error": "فشل التحليل"}), 500

async function registerForPushNotificationsAsync() {
  let token;
  const { status: existingStatus } = await Notifications.getPermissionsAsync();
  let finalStatus = existingStatus;
  if (existingStatus !== 'granted') {
    const { status } = await Notifications.requestPermissionsAsync();
    finalStatus = status;
  }
  if (finalStatus !== 'granted') {
    alert('فشل الحصول على إذن للإشعارات!');
    return;
  }
  token = (await Notifications.getExpoPushTokenAsync()).data;
  console.log("Expo Push Token:", token);

  if (Platform.OS === 'android') {
    Notifications.setNotificationChannelAsync('default', {
      name: 'default',
      importance: Notifications.AndroidImportance.MAX,
      vibrationPattern: [0, 250, 250, 250],
      lightColor: '#FF231F7C',
    });
  }

  return token;
}

export default function App() {
  const [isReady, setIsReady] = useState(false);

  useEffect(() => {
    const bootstrapAsync = async () => {
      try {
        // استعادة التوكن عند بدء التشغيل
        const token = await AsyncStorage.getItem('userToken');
        if (token) {
          setAuthToken(token); // إعداد التوكن في axios
        }

        // تسجيل للحصول على توكن الإشعارات وإرساله للخادم
        const expoPushToken = await registerForPushNotificationsAsync();
        if (expoPushToken && token) { // تأكد من أن المستخدم مسجل الدخول
          try {
            // حفظ التوكن في تفضيلات المستخدم
             await api.put('/auth/profile', { preferences: { expo_push_token: expoPushToken } });
             console.log("Expo push token saved to profile");
          } catch (profileError) {
             console.error("Failed to save expo push token:", profileError);
          }
        }
      } catch (e) {
        console.warn(e);
      } finally {
        setIsReady(true);
      }
    };

    bootstrapAsync();

     // الاستماع للإشعارات الواردة
     const notificationListener = Notifications.addNotificationReceivedListener(notification => {
        console.log("Notification received:", notification);
     });

     // الاستماع لاستجابة المستخدم للإشعار
     const responseListener = Notifications.addNotificationResponseReceivedListener(response => {
        console.log("Notification response:", response);
        // يمكنك هنا التوجيه بناءً على بيانات الإشعار
     });

     return () => {
        Notifications.removeNotificationSubscription(notificationListener);
        Notifications.removeNotificationSubscription(responseListener);
     };

  }, []);

  if (!isReady) {
    return null; // أو عرض شاشة تحميل
  }

  return (
    <>
      <AppNavigator />
      <StatusBar style="auto" />
    </>
  );
}

export default function ARView({ modelUrl }) {
  const glRef = useRef<GLView | null>(null);

  useEffect(() => {
    return () => {
      // تنظيف الموارد عند إلغاء تحميل المكون
      if (glRef.current) {
        // أي عمليات تنظيف لـ THREE.js
      }
    };
  }, []);

  const onContextCreate = async (gl: ExpoWebGLRenderingContext) => {
    glRef.current = gl;
    const { drawingBufferWidth: width, drawingBufferHeight: height } = gl;
    const sceneColor = 0x6ec6ff;

    // إنشاء العارض
    const renderer = new Renderer({ gl });
    renderer.setSize(width, height);
    renderer.setClearColor(sceneColor);

    // إنشاء الكاميرا
    const camera = new PerspectiveCamera(70, width / height, 0.01, 1000);
    camera.position.set(0, 1, 5); // تعديل موضع الكاميرا

    // إنشاء المشهد
    const scene = new Scene();
    scene.fog = new THREE.Fog(sceneColor, 1, 10000);
    scene.add(new GridHelper(10, 10)); // إضافة شبكة للمساعدة البصرية

    // إضافة الإضاءة
    const ambientLight = new AmbientLight(0x101010);
    scene.add(ambientLight);
    const pointLight = new PointLight(0xffffff, 2, 1000, 1);
    pointLight.position.set(0, 200, 200);
    scene.add(pointLight);

    // تحميل نموذج GLTF
    try {
       const model = await loadGLTF(modelUrl); // استخدم دالة تحميل GLTF
       model.scene.scale.set(0.1, 0.1, 0.1); // تصغير النموذج
       model.scene.position.y = 0; // تعديل الموضع
       scene.add(model.scene);
       console.log("Model loaded successfully");
    } catch (e) {
       console.error("Error loading model:", e);
       Alert.alert("خطأ", "فشل تحميل نموذج AR.");
    }


    // وظيفة العرض
    const render = () => {
      requestAnimationFrame(render);
      // يمكنك إضافة دوران أو تفاعل هنا

      gl.endFrameEXP();

    render();

    </View>
  );
}
