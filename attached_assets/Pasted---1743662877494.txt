# Renaming for clarity, acknowledging limitations.
class SecureVault:
    def __init__(self):
        # Using Fernet for strong symmetric encryption. Key management is crucial.
        # For production, the key should be managed securely (e.g., Vault, KMS)
        # and not generated randomly on each instantiation.
        # This key generation is illustrative.
        key = os.environ.get("DATA_ENCRYPTION_KEY")
        if not key:
            logger.warning("DATA_ENCRYPTION_KEY not set, generating ephemeral key.")
            key = Fernet.generate_key()
        else:
            key = key.encode() # Ensure key is bytes
        self.cipher_suite = Fernet(key)
        # Note: Storing a hash derived from random bytes isn't standard practice for key derivation.
        # self.key_hash = hashlib.sha3_512(np.random.bytes(1024)).digest()

    def encrypt(self, text: str) -> str:
        """Encrypts text using Fernet."""
        try:
            encrypted_text = self.cipher_suite.encrypt(text.encode())
            return encrypted_text.decode()
        except Exception as e:
            logger.error(f"Encryption failed: {e}", exc_info=True)
            raise # Re-raise to indicate failure

    def decrypt(self, encrypted_text: str) -> str:
        """Decrypts text using Fernet."""
        try:
            decrypted_text = self.cipher_suite.decrypt(encrypted_text.encode())
            return decrypted_text.decode()
        except Exception as e: # Catch more specific errors like InvalidToken if needed
            logger.error(f"Decryption failed: {e}", exc_info=True)
            raise # Re-raise to indicate failure

def check_bias(text: str) -> bool:
    """Checks for potential bias terms in the text (simple check)."""
    bias_terms = ['gender', 'race', 'religion', 'sex', 'ethnicity'] # Expanded list
    return any(term in text.lower() for term in bias_terms)

def sanitize_input(text: str) -> str:
    """Basic input sanitization against common injection patterns."""
    # This is basic. Use libraries like Bleach for robust HTML sanitization if needed.
    blacklist = [';', '--', '/*', '*/', 'xp_', '<script>', '</script>', 'SELECT', 'INSERT', 'UPDATE', 'DELETE', 'DROP']
    sanitized_text = text
    for item in blacklist:
        sanitized_text = sanitized_text.replace(item, "") # Simple replacement
    return sanitized_text

def validate_jwt(token: str):
    """Validates the JWT token."""
    try:
        decoded = jwt.decode(
            token,
            app.secret_key, # Use the app's secret key
            algorithms=['HS512'],
            issuer='BTEC_REBEL_SYSTEM', # Verify issuer
            options={'verify_iss': True, 'require': ['exp', 'user']} # Require essential claims
        )
        return decoded['user']
    except jwt.ExpiredSignatureError:
        logger.warning("JWT validation failed: Token expired")
        return "TOKEN_EXPIRED"
    except jwt.InvalidIssuerError:
        logger.warning("JWT validation failed: Invalid issuer")
        return "INVALID_ISSUER"
    except jwt.MissingRequiredClaimError as e:
         logger.warning(f"JWT validation failed: Missing claim {e}")
         return "INVALID_TOKEN"
    except jwt.InvalidTokenError as e:
        logger.warning(f"JWT validation failed: Invalid token - {e}")
        return "INVALID_TOKEN"

@app.route('/api/v1/register', methods=['POST'])
@limiter.limit("5 per hour") # Limit registration attempts
def register():
    """Registers a new user."""
    data = request.get_json()
    if not data or 'email' not in data or 'password' not in data or 'role' not in data:
        return jsonify({'error': 'Missing email, password, or role'}), 400

    email = sanitize_input(data['email'])
    password = data['password'] # Don't sanitize password before hashing
    role = sanitize_input(data['role'])

    if role not in ['admin', 'evaluator', 'student']:
        return jsonify({'error': 'Invalid role specified'}), 400

    # Hash the password securely
    hashed_pass = generate_password_hash(password)

    conn = None
    try:
        conn = get_db_conn()
        c = conn.cursor()
        c.execute("INSERT INTO users (email, password_hash, role) VALUES (?, ?, ?)",
                  (email, hashed_pass, role))
        conn.commit()
        log_audit("User Registration", "System", f"User {email} registered with role {role}")
        logger.info(f"User {email} registered successfully")
        return jsonify({'message': f'User {email} registered successfully'}), 201
    except sqlite3.IntegrityError: # Catch duplicate email error
        logger.warning(f"Registration failed: Email {email} already exists.")
        return jsonify({'error': f'Email {email} already exists'}), 409
    except sqlite3.Error as e:
        logger.error(f"Database error during registration: {e}", exc_info=True)
        return jsonify({'error': 'Registration failed due to database error'}), 500
    finally:
        if conn:
            conn.close()


@app.route('/api/v1/login', methods=['POST'])
@limiter.limit("10 per minute") # Limit login attempts
def login():
    """Authenticates a user and returns a JWT."""
    data = request.get_json()
    if not data or 'email' not in data or 'password' not in data:
        return jsonify({'error': 'Missing email or password'}), 400

    email = sanitize_input(data['email'])
    password = data['password']

    conn = None
    try:
        conn = get_db_conn()
        c = conn.cursor()
        c.execute("SELECT password_hash, role FROM users WHERE email = ?", (email,))
        user_record = c.fetchone()

        if user_record and check_password_hash(user_record['password_hash'], password):
            # Password matches, generate JWT
            token = jwt.encode({
                'user': email,
                'role': user_record['role'], # Include role in token
                'exp': datetime.datetime.utcnow() + datetime.timedelta(hours=2),
                'iss': 'BTEC_REBEL_SYSTEM' # Issuer
            }, app.secret_key, algorithm='HS512')
            log_audit("User Login", email, "Successful login")
            logger.info(f"User {email} logged in successfully")
            return jsonify({'token': token})
        else:
            log_audit("User Login Failed", email, "Invalid credentials")
            logger.warning(f"Login failed for user {email}: Invalid credentials")
            return jsonify({'error': 'Invalid email or password'}), 401

    except sqlite3.Error as e:
        logger.error(f"Database error during login: {e}", exc_info=True)
        return jsonify({'error': 'Login failed due to database error'}), 500
    finally:
        if conn:
            conn.close()

from functools import wraps

def token_required(f):
    @wraps(f)
    def decorated(*args, **kwargs):
        token = None
        if 'Authorization' in request.headers:
            auth_header = request.headers['Authorization']
            try:
                token = auth_header.split(" ")[1]
            except IndexError:
                return jsonify({'message': 'Bearer token malformed'}), 401

        if not token:
            return jsonify({'message': 'Token is missing'}), 401

        user_email = validate_jwt(token)
        if user_email in ["TOKEN_EXPIRED", "INVALID_TOKEN", "INVALID_ISSUER"]:
             return jsonify({'message': f'Token error: {user_email}'}), 401

        # You might want to fetch user details from DB here if needed
        # current_user = get_user_details(user_email)
        # g.current_user = current_user # Store user in Flask's 'g' object

        return f(user_email, *args, **kwargs) # Pass user email to the route function
    return decorated

4. Security Implementation
4.1 آليات الدفاع

التشفير القوي (Fernet):

يستخدم تشفير Fernet المتماثل القوي.

إدارة المفاتيح: تعتمد أمان Fernet بشكل كبير على إدارة المفتاح السري (DATA_ENCRYPTION_KEY و FLASK_SECRET_KEY). يجب تخزين هذه المفاتيح وإدارتها بشكل آمن للغاية باستخدام متغيرات البيئة أو أنظمة إدارة الأسرار (مثل HashiCorp Vault، AWS Secrets Manager، Azure Key Vault). لا تقم أبداً بتضمين المفاتيح مباشرة في الكود.

(مقترح) تنفيذ تدوير المفاتيح بشكل دوري لهذه المفاتيح.

التحقق من صحة JWT:

يستخدم JWT مع خوارزمية HS512.

يتحقق من المُصدر (iss) وتاريخ انتهاء الصلاحية (exp).

يتضمن دور المستخدم (role) في الحمولة للتحكم في الوصول المستند إلى الأدوار.

يستخدم app.secret_key لتوقيع والتحقق من الرموز.

تنظيف الإدخال:

تطبيق دالة sanitize_input أساسية لإزالة بعض الأنماط الشائعة لهجمات الحقن (مثل SQL injection و XSS). ملاحظة: للحماية القوية ضد XSS في المحتوى المعروض، استخدم مكتبات مثل Bleach.

تحديد المعدل (Rate Limiting):

تم التنفيذ باستخدام Flask-Limiter للحد من عدد الطلبات إلى نقاط النهاية المختلفة، مما يساعد في الحماية من هجمات القوة الغاشمة ورفض الخدمة البسيطة.

HTTPS:

التطبيق مهيأ للتشغيل عبر HTTPS باستخدام ssl_context، وهو أمر ضروري لتشفير البيانات أثناء النقل. تأكد من توفر شهادات صالحة (cert.pem, key.pem).

التجزئة الآمنة لكلمات المرور:

يستخدم generate_password_hash و check_password_hash من Werkzeug، والتي تطبق تجزئة مملحة (salting) وتكرارات متعددة (مثل PBKDF2) لتخزين كلمات المرور بشكل آمن.

4.2 نموذج التهديد (أمثلة)

حقن SQL: على الرغم من وجود تنظيف أساسي، قد تظل الاستعلامات المعقدة أو التي تم إنشاؤها ديناميكيًا عرضة للخطر إذا لم يتم استخدام استعلامات معلمة (parameterized queries) بشكل صحيح (الكود الحالي يستخدمها بشكل جيد مع ?).

Cross-Site Scripting (XSS): إذا تم عرض الملاحظات التي تم إنشاؤها بواسطة AI أو أي إدخال مستخدم آخر مباشرة في واجهة أمامية دون تطهير مناسب من جانب الخادم أو العميل.

الوصول غير المصرح به: اختراق رموز JWT، ضعف التحكم في الوصول المستند إلى الأدوار، نقاط نهاية غير محمية.

تسرب البيانات: اختراق قاعدة البيانات، ضعف تشفير البيانات الثابتة (إذا لم تتم إدارة المفتاح بشكل صحيح)، أخطاء في التسجيل تكشف عن بيانات حساسة.

هجمات رفض الخدمة (DoS/DDoS): إغراق الخادم بالطلبات، استغلال نقاط النهاية كثيفة الاستخدام للموارد.

مشاكل التحكم في الوصول: السماح للمستخدمين بتنفيذ إجراءات تتجاوز أدوارهم المحددة (مثل السماح لطالب بتشغيل self_destruct).

أخطاء التكوين: كشف المفاتيح السرية، أذونات غير آمنة للملفات/الخدمات، استخدام برامج قديمة.

4.3 تدابير أمنية إضافية موصى بها

سياسة أمان المحتوى (CSP): قم بتكوين رؤوس CSP في خادم الويب (مثل Nginx) لتقليل مخاطر XSS.

رؤوس الأمان: أضف رؤوس أمان أخرى مثل X-Content-Type-Options: nosniff, X-Frame-Options: DENY, Strict-Transport-Security (HSTS).

نظام إدارة الأسرار: استخدم نظامًا مخصصًا لإدارة المفاتيح والبيانات السرية بدلاً من الاعتماد فقط على متغيرات البيئة.

تدقيق أمني منتظم: قم بإجراء اختبارات اختراق وتدقيقات أمنية دورية بواسطة خبراء.

مراقبة أمنية: استخدم أدوات لمراقبة السجلات بحثًا عن الأنشطة المشبوهة والتنبيه بشأنها.

Web Application Firewall (WAF): ضع WAF أمام التطبيق لتصفية حركة المرور الضارة.

التحكم الصارم في الوصول المستند إلى الأدوار: تحقق دائمًا من دور المستخدم (من JWT أو قاعدة البيانات) قبل السماح بالوصول إلى العمليات الحساسة.

7. المبادئ التوجيهية الأخلاقية
7.1 حقوق الطالب

الشفافية: الحق في فهم كيف يتم استخدام الذكاء الاصطناعي في عملية التقييم وما هي المعايير المطبقة.

التصحيح: الحق في طلب تصحيح أي بيانات شخصية أو تقييمية غير دقيقة.

المراجعة البشرية: الحق في طلب مراجعة بشرية لتقييم تم إنشاؤه بواسطة الذكاء الاصطناعي، خاصة في حالة وجود اعتراضات أو نتائج غير متوقعة.

الخصوصية: الحق في معرفة كيفية جمع بياناتهم واستخدامها وحمايتها، بما يتوافق مع لوائح حماية البيانات.

7.2 تخفيف التحيز ومراقبته

مطالبات (Prompts) واعية بالتحيز: تصميم مطالبات GPT-4 لتوجيه النموذج بشكل صريح للبحث عن التحيزات المحتملة وتجنبها في تقييماته.

فحص المخرجات: تنفيذ آليات (مثل check_bias أو تقنيات أكثر تقدماً لتحليل المشاعر/المواضيع) لفحص الملاحظات التي تم إنشاؤها بحثًا عن لغة متحيزة. وضع علامة على التقييمات التي يحتمل أن تكون متحيزة للمراجعة البشرية.

بيانات الضبط الدقيق المتنوعة (إذا تم استخدامه): إذا تم استخدام الضبط الدقيق، فتأكد من أن مجموعة بيانات التدريب متنوعة وتمثل مجموعات طلابية مختلفة لتجنب ترميز التحيزات الموجودة.

المراقبة المستمرة: تحليل نتائج التقييم بشكل دوري عبر مجموعات ديموغرافية مختلفة (إذا كانت هذه البيانات متاحة بشكل أخلاقي وقانوني) لتحديد أي أنماط تحيز نظامية قد تظهر بمرور الوقت.

حلقة التغذية الراجعة: إنشاء آلية للطلاب والمعلمين لتقديم ملاحظات حول جودة وعدالة التقييمات التي تم إنشاؤها بواسطة الذكاء الاصطناعي، واستخدام هذه الملاحظات لتحسين النظام.

7.3 الرقابة البشرية

إجراءات المراجعة: تحديد إجراءات واضحة للمراجعة البشرية للتقييمات، خاصة للتقييمات عالية المخاطر، أو التقييمات التي تم وضع علامة عليها لوجود تحيز محتمل، أو بناءً على طلب الطالب/المعلم.

دور المعلم: يجب أن يظل المعلم هو المقرر النهائي للدرجات، مع استخدام تقييم الذكاء الاصطناعي كأداة مساعدة لتقديم ملاحظات تفصيلية ومتسقة، وليس كبديل كامل للحكم البشري.

التدريب: توفير التدريب للمعلمين حول كيفية استخدام النظام بفعالية، وتفسير مخرجات الذكاء الاصطناعي، وتطبيق الرقابة البشرية.

13. تحليلات متقدمة

تحليل ملاحظات التقييم: يمكن استخدام نقطة نهاية analyze_data (كما في الكود) لتحليل المشاعر أو استخراج المواضيع الرئيسية من الملاحظات التي تم إنشاؤها بواسطة الذكاء الاصطناعي لتقييم جودتها أو تحديد الأنماط.

تحليل أداء الطلاب: (يتطلب تصميم قاعدة بيانات أكثر تفصيلاً) تحليل الدرجات المشفرة (بعد فك تشفيرها في بيئة آمنة ومراقبة) لتتبع تقدم الطلاب وتحديد المجالات التي قد تحتاج إلى دعم إضافي. يجب أن يتم ذلك مع الامتثال الصارم للخصوصية والأخلاقيات.

تحليل أداء النموذج: مراقبة مقاييس مثل درجة التحيز المحسوبة، طول الملاحظات، أو استخدام الرموز (tokens) لتقييم أداء نموذج GPT-4 بمرور الوقت.

14. المراقبة والصيانة
14.1 مقاييس الأداء والتطبيق (مثال Prometheus/Grafana)

مقاييس التطبيق: استخدم مكتبة مثل prometheus-flask-exporter لكشف مقاييس مثل زمن استجابة الطلب، عدد الطلبات، عدد الأخطاء (HTTP 5xx, 4xx).

مقاييس النظام: مراقبة استخدام وحدة المعالجة المركزية (CPU)، والذاكرة (RAM)، والقرص (Disk I/O)، والشبكة للخوادم أو الحاويات.

مقاييس قاعدة البيانات: مراقبة أداء استعلامات قاعدة البيانات، عدد الاتصالات، حجم قاعدة البيانات.

مقاييس واجهة برمجة تطبيقات OpenAI: مراقبة زمن الاستجابة، ومعدلات الأخطاء، واستخدام الرموز لواجهة برمجة تطبيقات OpenAI.

14.2 تحليل السجلات المركزي

التجميع: قم بإرسال سجلات التطبيق وسجلات خادم الويب وسجلات النظام إلى نظام تسجيل مركزي (مثل ELK Stack - Elasticsearch, Logstash, Kibana؛ أو Splunk؛ أو Graylog).

التحليل والتنبيه: قم بتكوين لوحات معلومات لتحليل السجلات وتحديد الاتجاهات أو المشكلات. قم بإعداد تنبيهات لأنماط السجل الهامة (مثل الأخطاء الفادحة، محاولات تسجيل الدخول الفاشلة المتعددة، اكتشاف التحيز المحتمل).

14.3 فحوصات السلامة

استخدم نقطة النهاية /health التي تم تنفيذها للتحقق من حالة التطبيق واتصال قاعدة البيانات. يمكن استخدام هذا بواسطة أنظمة المراقبة وموازنات التحميل و Kubernetes (probes).
