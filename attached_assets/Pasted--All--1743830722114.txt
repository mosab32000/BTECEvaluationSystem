let currentPdfDoc = null;
let currentPageNum = 1;
let currentScale = 1.5;
const pdfViewerCanvas = document.getElementById('pdf-viewer');
const pdfPageNumSpan = document.getElementById('pdf-page-num');
const pdfPageCountSpan = document.getElementById('pdf-page-count');
const pdfPrevButton = document.getElementById('pdf-prev');
const pdfNextButton = document.getElementById('pdf-next');
const pdfZoomInButton = document.getElementById('pdf-zoom-in');
const pdfZoomOutButton = document.getElementById('pdf-zoom-out');
const pdfViewerContainer = document.getElementById('pdf-viewer-container');

function renderPage(num, scale) {
    if (!currentPdfDoc) return;
    currentPdfDoc.getPage(num).then(page => {
        const viewport = page.getViewport({ scale: scale });
        const canvasContext = pdfViewerCanvas.getContext('2d');
        pdfViewerCanvas.height = viewport.height;
        pdfViewerCanvas.width = viewport.width;

        const renderContext = { canvasContext, viewport };
        page.render(renderContext).promise.then(() => {
            currentPageNum = num;
            currentScale = scale;
            pdfPageNumSpan.textContent = currentPageNum;
            pdfPrevButton.disabled = (currentPageNum <= 1);
            pdfNextButton.disabled = (currentPageNum >= currentPdfDoc.numPages);
        }).catch(err => {
            showMessage(`فشل عرض الصفحة ${num}`, 'error');
        });
    });
}

function renderNextPage() {
    if (currentPdfDoc && currentPageNum < currentPdfDoc.numPages) {
        renderPage(currentPageNum + 1, currentScale);
    }
}

function renderPreviousPage() {
    if (currentPdfDoc && currentPageNum > 1) {
        renderPage(currentPageNum - 1, currentScale);
    }
}

function zoomInPDF() {
    if (!currentPdfDoc || currentScale >= 3.0) return;
    renderPage(currentPageNum, currentScale + 0.25);
}

function zoomOutPDF() {
    if (!currentPdfDoc || currentScale <= 0.5) return;
    renderPage(currentPageNum, currentScale - 0.25);
}

function openPdfViewer(pdfUrl) {
    pdfViewerContainer.style.display = 'block';
    pdfViewerContainer.querySelector('canvas').style.display = 'none';
    pdfViewerContainer.insertAdjacentHTML('beforeend', '<p id="pdf-loading-msg">جاري تحميل المخطوطة...</p>');

    pdfjsLib.getDocument(pdfUrl).promise.then(pdfDoc_ => {
        currentPdfDoc = pdfDoc_;
        pdfPageCountSpan.textContent = currentPdfDoc.numPages;
        currentPageNum = 1;
        currentScale = 1.5;
        document.getElementById('pdf-loading-msg').remove();
        pdfViewerContainer.querySelector('canvas').style.display = 'block';
        renderPage(currentPageNum, currentScale);
    }).catch(err => {
        showMessage('فشل تحميل المخطوطة!', 'error');
    });
}

function closePdfViewer() {
    pdfViewerContainer.style.display = 'none';
    currentPdfDoc = null;
}

const messageBox = document.getElementById('message-box');
let messageTimeout;

function showMessage(message, type = 'info') {
    if (!messageBox) return;
    messageBox.textContent = message;
    messageBox.classList.remove('success', 'error', 'info', 'show');
    setTimeout(() => {
        messageBox.classList.add(type, 'show');
    }, 10);
    clearTimeout(messageTimeout);
    messageTimeout = setTimeout(() => {
        messageBox.classList.remove('show');
    }, 5000);
}

from rest_framework.decorators import api_view, permission_classes
from rest_framework.response import Response
from rest_framework import status
from rest_framework.permissions import IsAuthenticated
from django.shortcuts import get_object_or_404
from django.utils import timezone
from .models import Class, AttendanceLog
from .serializers import BulkAttendanceSerializer

@api_view(['POST'])
@permission_classes([IsAuthenticated])
def bulk_mark_attendance(request, class_id):
    class_group = get_object_or_404(Class, id=class_id)
    if not hasattr(request.user, 'is_teacher') or not request.user.is_teacher or class_group.teacher != request.user:
        return Response({'error': 'غير مصرح لك بأخذ الحضور لهذا الفصل'}, status=status.HTTP_403_FORBIDDEN)

    serializer = BulkAttendanceSerializer(data=request.data)
    if not serializer.is_valid():
        return Response({'error': 'بيانات غير صالحة', 'details': serializer.errors}, status=status.HTTP_400_BAD_REQUEST)

    attendance_data = serializer.validated_data['attendance']
    today = timezone.now().date()
    results = {'success': 0, 'failed': 0, 'errors': []}
    class_student_ids = list(class_group.students.values_list('id', flat=True))

    for item in attendance_data:
        student_id = item['student_id']
        status_value = item['status']
        if student_id not in class_student_ids:
            results['failed'] += 1
            results['errors'].append(f"Student ID {student_id} غير مسجل في هذا الفصل.")
            continue
        try:
            log, created = AttendanceLog.objects.update_or_create(
                student_id=student_id,
                class_group=class_group,
                date=today,
                defaults={'status': status_value}
            )
            results['success'] += 1
        except Exception as e:
            results['failed'] += 1
            results['errors'].append(f"فشل معالجة ID {student_id}: {str(e)}")

    if results['failed'] > 0:
        return Response({
            'message': f"تمت معالجة الحضور بشكل جزئي. النجاح: {results['success']}, الفشل: {results['failed']}",
            'errors': results['errors']
        }, status=status.HTTP_207_MULTI_STATUS)
    return Response({'message': f"تم تسجيل حضور {results['success']} طالب بنجاح."}, status=status.HTTP_200_OK)
